-- APÊ COMPRAS V26 -- CORREÇÃO DE EXCLUSÃO DE CASAS E USUÁRIOS -- O histórico pode permanecer mesmo depois que o usuário é excluído. ALTER TABLE public.compras ALTER COLUMN usuario_id DROP NOT NULL; -- Execute este arquivo inteiro no SQL Editor do Supabase. CREATE OR REPLACE FUNCTION public.excluir_casa(p_casa_id uuid) RETURNS boolean LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ BEGIN -- Proprietário da casa ou administrador principal IF NOT public.usuario_atual_e_admin() AND NOT EXISTS ( SELECT 1 FROM public.membros_casa WHERE casa_id = p_casa_id AND usuario_id = auth.uid() AND papel = 'proprietario' ) THEN RAISE EXCEPTION 'Sem permissão para excluir esta casa.'; END IF; -- Primeiro remove os itens das compras. -- Isso evita o erro de chave estrangeira produtos/itens_compra. DELETE FROM public.itens_compra WHERE compra_id IN ( SELECT id FROM public.compras WHERE casa_id = p_casa_id ); -- Depois remove o histórico de compras. DELETE FROM public.compras WHERE casa_id = p_casa_id; -- Agora os produtos não possuem mais itens de compra vinculados. DELETE FROM public.produtos WHERE casa_id = p_casa_id; -- Remove os membros/vínculos da casa. DELETE FROM public.membros_casa WHERE casa_id = p_casa_id; -- Por último remove a própria casa. DELETE FROM public.casas WHERE id = p_casa_id; IF NOT FOUND THEN RAISE EXCEPTION 'Casa não encontrada.'; END IF; RETURN true; END; $$; GRANT EXECUTE ON FUNCTION public.excluir_casa(uuid) TO authenticated; CREATE OR REPLACE FUNCTION public.excluir_casa_admin(p_casa_id uuid) RETURNS boolean LANGUAGE plpgsql SECURITY DEFINER SET search_path = public AS $$ BEGIN IF NOT public.usuario_atual_e_admin() THEN RAISE EXCEPTION 'Acesso permitido somente ao administrador principal.'; END IF; DELETE FROM public.itens_compra WHERE compra_id IN ( SELECT id FROM public.compras WHERE casa_id = p_casa_id ); DELETE FROM public.compras WHERE casa_id = p_casa_id; DELETE FROM public.produtos WHERE casa_id = p_casa_id; DELETE FROM public.membros_casa WHERE casa_id = p_casa_id; DELETE FROM public.casas WHERE id = p_casa_id; IF NOT FOUND THEN RAISE EXCEPTION 'Casa não encontrada.'; END IF; RETURN true; END; $$; GRANT EXECUTE ON FUNCTION public.excluir_casa_admin(uuid) TO authenticated;
